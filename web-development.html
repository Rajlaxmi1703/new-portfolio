My JavaScript Code
function addTodo() {
  const input = document.getElementById('todoInput');
  const value = input.value.trim();
  if (!value) return;

  const item = document.createElement('div');
  item.className = 'todo-item';
  item.onclick = function() { toggleTodo(this); };
  item.innerHTML = `
    <i class="fas fa-circle"></i>
    <span>${value}</span>
  `;
  document.getElementById('todoItems').appendChild(item);
  input.value = '';
}

function toggleTodo(item) {
  const icon   = item.querySelector('i');
  const isDone = icon.classList.contains('fa-check-circle');
  icon.classList.toggle('fa-check-circle', !isDone);
  icon.classList.toggle('fa-circle',       isDone);
}

💡 What I learned: JavaScript makes pages interactive! DOM manipulation, event handling, dynamic elements, and building a To-Do List app.
